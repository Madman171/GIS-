def load_snapshot(path: Path) -> Dict[str, Any]:
    snapshot = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(snapshot.get("nodes"), list):
        raise ValueError("快照缺少 nodes 数组")
    if not isinstance(snapshot.get("relationships"), list):
        raise ValueError("快照缺少 relationships 数组")
    return snapshot


def node_business_id(node: Dict[str, Any]) -> str:
    label = str(node.get("primaryLabel", ""))
    key = PRIMARY_ID_KEYS.get(label)
    properties = node.get("properties", {})
    if key and properties.get(key) is not None:
        return str(properties[key])
    if node.get("businessId") is not None:
        return str(node["businessId"])
    return str(node.get("elementId", ""))

def _validate_coordinate(properties: Dict[str, Any]) -> Optional[bool]:
    has_x = properties.get("x") is not None or properties.get("lon") is not None
    has_y = properties.get("y") is not None or properties.get("lat") is not None
    if not has_x and not has_y:
        return None
    try:
        x_value = properties.get("x", properties.get("lon"))
        y_value = properties.get("y", properties.get("lat"))
        x = float(x_value)
        y = float(y_value)
        return -180 <= x <= 180 and -90 <= y <= 90
    except (TypeError, ValueError):
        return False

class TrafficGraphBuilder:
    def __init__(self) -> None:
        self.nodes: List[Dict[str, Any]] = []
        self.links: List[Dict[str, Any]] = []
        self._node_ids: Set[str] = set()
        self._link_ids: Set[str] = set()

    def add_node(self, node: Optional[Node]) -> None:
        if node is None:
            return
        serialized = _serialize_node(node)
        if serialized["id"] in self._node_ids:
            return
        self._node_ids.add(serialized["id"])
        self.nodes.append(serialized)

    def add_relationship(self, relationship: Optional[Relationship]) -> None:
        if relationship is None:
            return
        serialized = _serialize_relationship(relationship)
        if serialized["id"] in self._link_ids:
            return
        self._link_ids.add(serialized["id"])
        self.links.append(serialized)

    def as_dict(self) -> Dict[str, Any]:
        return {
            "nodes": self.nodes,
            "links": self.links,
        }

def _check_chat_rate_limit(request: Request) -> None:
    client_ip = request.headers.get(
        "x-real-ip",
        request.client.host if request.client else "unknown",
    )
    if "*" in QA_CONFIG["whitelist"] or client_ip in QA_CONFIG["whitelist"]:
        return

    now = time.time()
    window_start = now - 60

    with CHAT_RATE_LOCK:
        request_times = [
            timestamp
            for timestamp in CHAT_REQUEST_TIMES.get(client_ip, [])

def collect_source_files() -> List[Path]:
    files = set()
    for pattern in SOURCE_PATTERNS:
        for path in ROOT_DIR.glob(pattern):
            if not path.is_file():
                continue
            relative = path.relative_to(ROOT_DIR)
            if any(part in EXCLUDED_PARTS for part in relative.parts):
                continue
            if path.name in EXCLUDED_FILES:
                continue
            files.add(path)

    return sorted(
        files,
        key=lambda path: path.relative_to(ROOT_DIR).as_posix().lower(),
    )

def cypher_literal(value: Any) -> str:
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "true" if value else "false"
    if isinstance(value, (int, float)):
        return str(value)
    if isinstance(value, str):
        return json.dumps(value, ensure_ascii=False)
    if isinstance(value, (list, tuple)):
        return "[" + ", ".join(cypher_literal(item) for item in value) + "]"
    if isinstance(value, dict):
        entries = ", ".join(
            "{0}: {1}".format(
                "`{0}`".format(str(key).replace("`", "``")),
                cypher_literal(item),
            )
            for key, item in value.items()
        )

def validate_graph(
    nodes: List[Dict[str, Any]],
    relationships: List[Dict[str, Any]],
) -> None:
    node_element_ids = set()
    business_ids = set()

    for node in nodes:
        if node["elementId"] in node_element_ids:
            raise ValueError(
                "Duplicate node element ID: {0}".format(node["elementId"])
            )
        node_element_ids.add(node["elementId"])

        if not node["businessId"]:
            raise ValueError(
                "Missing business ID for node {0}".format(node["elementId"])
            )
        business_key = (node["primaryLabel"], node["businessId"])
        if business_key in business_ids:
            raise ValueError(
                "Duplicate business ID: {0}".format(business_key)
            )
        business_ids.add(business_key)

    relationship_signatures = set()
    for relationship in relationships:
        if relationship["sourceElementId"] not in node_element_ids:
            raise ValueError("Relationship source is missing from nodes")
        if relationship["targetElementId"] not in node_element_ids:
            raise ValueError("Relationship target is missing from nodes")
        if not relationship["sourceBusinessId"]:
            raise ValueError("Relationship source business ID is missing")
        if not relationship["targetBusinessId"]:
            raise ValueError("Relationship target business ID is missing")

        signature = (
            relationship["sourceBusinessId"],
            relationship["type"],
            relationship["targetBusinessId"],
        )
        if signature in relationship_signatures:
            raise ValueError(
                "Duplicate relationship signature: {0}".format(signature)
            )
        relationship_signatures.add(signature)
